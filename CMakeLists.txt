cmake_minimum_required(VERSION 3.16)
project(CPlusScanner VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(cisscanner-agent agent/main.cpp)
target_include_directories(cisscanner-agent PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
add_executable(cisscanner-controller controller/main.cpp)
target_include_directories(cisscanner-controller PRIVATE ${CMAKE_CURRENT_SOURCE_DIR}/include)
if(MSVC)
  target_compile_options(cisscanner-agent PRIVATE /W4 /permissive-)
  target_compile_options(cisscanner-controller PRIVATE /W4 /permissive-)
else()
  target_compile_options(cisscanner-agent PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(cisscanner-controller PRIVATE -Wall -Wextra -Wpedantic)
endif()
