cmake_minimum_required(VERSION 3.16)
project(CPlusScanner VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_executable(cisscanner-agent agent/main.cpp)
target_include_directories(cisscanner-agent PRIVATE include)
add_executable(cisscanner-controller controller/main.cpp)
target_include_directories(cisscanner-controller PRIVATE include)
if(MSVC)
  target_compile_options(cisscanner-agent PRIVATE /W4 /permissive-)
  target_compile_options(cisscanner-controller PRIVATE /W4 /permissive-)
else()
  target_compile_options(cisscanner-agent PRIVATE -Wall -Wextra -Wpedantic)
  target_compile_options(cisscanner-controller PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME agent_ping COMMAND cisscanner-agent --ping --request-id cmake-test)
add_test(NAME agent_system_info COMMAND cisscanner-agent --system-info --request-id cmake-test)
